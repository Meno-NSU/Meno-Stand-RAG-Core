# S2 — Feedback Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add per-answer 👍/👎 (+ optional comment) and an end-of-session survey, stored in two new tables that reference turns by `run_id` (no content duplication), exposed via a `/v1/feedback` router, and surfaced in the analytics export.

**Architecture:** Two greenfield tables (`message_feedback`, `session_surveys`), each with `session_id` + nullable `user_id` (S3 hook). A new `feedback.py` router mirrors `arena.py`. Repository helpers upsert by `(run_id, session_id)` / `session_id`. Feedback references `run_id` as a **soft** indexed string (no hard FK) so a vote is never rejected when S1's best-effort persistence missed the run. The S1 analytics export gains a LEFT JOIN to feedback.

**Tech Stack:** FastAPI, SQLAlchemy 2.x async, Alembic, pydantic, pytest.

**Branch:** `claude/feedback-s2` (already checked out, off `main` with S0+S1 merged).

**Commit convention:** every commit message ends with the trailer shown in Task 1 Step 5.

```python
async def upsert_message_feedback(
    session: AsyncSession,
    *,
    run_id: str,
    session_id: str,
    value: str,
    comment: str | None = None,
    user_id: str | None = None,
) -> None:
    result = await session.execute(
        select(MessageFeedback).where(
            MessageFeedback.run_id == run_id,
            MessageFeedback.session_id == session_id,
        )
    )
    feedback = result.scalar_one_or_none()
    if feedback is None:
        session.add(
            MessageFeedback(
                run_id=run_id,
                session_id=session_id,
                value=value,
                comment=comment,
                user_id=user_id,
            )
        )
    else:
        feedback.value = value
        feedback.comment = comment
        if user_id is not None:
            feedback.user_id = user_id
        feedback.updated_at = datetime.now(UTC)


async def clear_message_feedback(session: AsyncSession, *, run_id: str, session_id: str) -> int:
    result = await session.execute(
        delete(MessageFeedback).where(
            MessageFeedback.run_id == run_id,
            MessageFeedback.session_id == session_id,
        )
    )
    return result.rowcount or 0


async def upsert_session_survey(
    session: AsyncSession,
    *,
    session_id: str,
    answer: str,
    user_id: str | None = None,
) -> None:
    result = await session.execute(select(SessionSurvey).where(SessionSurvey.session_id == session_id))
    survey = result.scalar_one_or_none()
    if survey is None:
        session.add(SessionSurvey(session_id=session_id, answer=answer, user_id=user_id))
    else:
        survey.answer = answer
        if user_id is not None:
            survey.user_id = user_id
        survey.updated_at = datetime.now(UTC)
```

def iter_analytics(session: Session, *, session_id: str | None = None) -> Iterator[dict]:
    stmt = (
        select(GenerationRecord, PipelineRun, MessageFeedback)
        .join(PipelineRun, GenerationRecord.run_id == PipelineRun.id)
        .outerjoin(MessageFeedback, MessageFeedback.run_id == PipelineRun.id)
        .order_by(PipelineRun.created_at)
    )
    if session_id is not None:
        stmt = stmt.where(PipelineRun.session_id == session_id)
    for gen, run, feedback in session.execute(stmt).all():
        yield {
            "run_id": run.id,
            "session_id": run.session_id,
            "created_at": run.created_at.isoformat() if run.created_at else None,
            "question": run.user_question,
            "search_queries": run.search_queries,
            "total_ms": run.total_ms,
            "response_len": run.response_len,
            "model": run.generation_model,
            "retrieved": gen.retrieved,
            "fewshots": gen.fewshots,
            "generation_params": gen.generation_params,
            "feedback": {"value": feedback.value, "comment": feedback.comment} if feedback is not None else None,
        }
